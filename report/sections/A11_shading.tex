\section{Shading: flat, Gouraud, Phong}
\label{sec:shading}
\here{8}
\concept{Shading decides where Eq.~\eqref{eq:phong} is evaluated. Flat: once per triangle with the face normal.
Gouraud: at the vertices, colours interpolated. Phong: normals interpolated and the model evaluated per fragment.}
\begin{equation}
\text{Gouraud: } I(\vect x)=\textstyle\sum_k\lambda_k\,I(\vect N_k),\qquad
\text{Phong: } I(\vect x)=I\Big(\frac{\sum_k\lambda_k\vect N_k}{\lVert\sum_k\lambda_k\vect N_k\rVert}\Big)
\label{eq:shading}
\end{equation}
